module mux4_1(input [1:0] sel, input [3:0] in, output out);
  wire s0_n, s1_n, y0, y1, y2, y3;
  
  assign s0_n = ~sel[0];
  assign s1_n = ~sel[1];
  
  assign y0 = in[0] & s1_n & s0_n;
  assign y1 = in[1] & s1_n &  sel[0];
  assign y2 = in[2] &  sel[1] & s0_n;
  assign y3 = in[3] &  sel[1] &  sel[0];

  assign out = y0 | y1 | y2 | y3;
endmodule
